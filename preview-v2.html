代码示例

Python 斐波那契数列：

import numpy as np
from typing import List

def fibonacci(n: int) -> List[int]:
    """生成斐波那契数列的前 n 项"""
    if n <= 0: return []
    if n == 1: return [0]
    fib = [0, 1]
    for i in range(2, n):
        fib.append(fib[i-1] + fib[i-2])
    return fib

print(fibonacci(10))
